<script>
  const API = '/api';
  let token = null;
  let user = null;
  let currentPage = 'dashboard';
  let sidebarOpen = false;
  let amcFilterStatus = 'all';
  let amcSearchQuery = '';

  function toggleSidebar() { sidebarOpen = !sidebarOpen; updateSidebar(); }
  function closeSidebar() { sidebarOpen = false; updateSidebar(); }
  function updateSidebar() {
    const sidebar = document.querySelector('.sidebar');
    const overlay = document.querySelector('.sidebar-overlay');
    if (sidebar) sidebar.classList.toggle('open', sidebarOpen);
    if (overlay) overlay.classList.toggle('open', sidebarOpen);
  }

  async function fetchAPI(endpoint, options = {}) {
    const headers = { 'Content-Type': 'application/json' };
    if (token) headers['Authorization'] = `Bearer ${token}`;
    const res = await fetch(API + endpoint, { ...options, headers });
    if (res.status === 401) { logout(); throw new Error('Session expired'); }
    if (!res.ok) throw new Error((await res.json()).error || 'Request failed');
    return res.json();
  }

  async function login(email, password) {
    try {
      const res = await fetch(API + '/auth/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password }) });
      if (!res.ok) throw new Error('Invalid email or password');
      const data = await res.json();
      token = data.token; user = data.user;
      localStorage.setItem('token', token); localStorage.setItem('user', JSON.stringify(user));
      renderApp();
    } catch (err) { document.getElementById('errorMsg') && (document.getElementById('errorMsg').innerText = err.message); }
  }

  async function signup(email, password, company_name, currency) {
    try {
      const res = await fetch(API + '/auth/signup', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ email, password, company_name, currency }) });
      if (!res.ok) throw new Error((await res.json()).error || 'Signup failed');
      const data = await res.json();
      token = data.token; user = data.user;
      localStorage.setItem('token', token); localStorage.setItem('user', JSON.stringify(user));
      renderApp();
    } catch (err) { document.getElementById('errorMsg') && (document.getElementById('errorMsg').innerText = err.message); }
  }

  async function demoLogin() {
    try {
      const res = await fetch(API + '/auth/demo', { method: 'POST' });
      const data = await res.json();
      token = data.token; user = data.user;
      localStorage.setItem('token', token); localStorage.setItem('user', JSON.stringify(user));
      renderApp();
    } catch (err) { alert('Demo login failed: ' + err.message); }
  }

  function logout() { token = null; user = null; localStorage.clear(); renderApp(); }

  let showSignup = false;
  function toggleForm() { showSignup = !showSignup; renderApp(); }

  function generatePDFInvoice(client, description, amount) {
    const { jsPDF } = window.jspdf;
    const doc = new jsPDF();
    const sym = user?.currency === 'EUR' ? '€' : 'AED';
    doc.setFont("helvetica");
    doc.setFontSize(18); doc.text("MAINTENANCE FLOW PRO - INVOICE", 14, 20);
    doc.setFontSize(10); doc.text(`Date: ${new Date().toLocaleDateString()}`, 14, 30);
    doc.text(`Invoice #: INV-${Date.now()}`, 14, 36);
    doc.autoTable({ startY: 45, head: [["Description", `Amount (${sym})`]], body: [[description, amount]], theme: "striped", styles: { fontSize: 10 } });
    doc.text(`Customer: ${client}`, 14, doc.lastAutoTable.finalY + 10);
    doc.text(`Total Amount: ${sym} ${amount}`, 14, doc.lastAutoTable.finalY + 18);
    doc.text("Thank you for your business - Maintenance Flow", 14, doc.lastAutoTable.finalY + 28);
    doc.save(`Invoice_${client.replace(/\s/g, "_")}.pdf`);
  }

  // Helper function to calculate days until expiry
  function getDaysUntilExpiry(expiryDate) {
    if (!expiryDate) return null;
    const today = new Date();
    const expiry = new Date(expiryDate);
    const diffTime = expiry - today;
    const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));
    return diffDays;
  }

  // Helper function to get contract status
  function getContractStatus(expiryDate, currentStatus) {
    if (currentStatus === 'Expired') return 'expired';
    const daysLeft = getDaysUntilExpiry(expiryDate);
    if (daysLeft === null) return 'active';
    if (daysLeft < 0) return 'expired';
    if (daysLeft <= 30) return 'expiring';
    return 'active';
  }

  // Helper function to calculate progress percentage
  function getContractProgress(startDate, expiryDate) {
    if (!startDate || !expiryDate) return 0;
    const start = new Date(startDate);
    const end = new Date(expiryDate);
    const today = new Date();
    if (today < start) return 0;
    if (today > end) return 100;
    const total = end - start;
    const elapsed = today - start;
    return Math.round((elapsed / total) * 100);
  }

  window.logAmcVisit = async (id) => {
    const amcs = await fetchAPI('/amcs');
    const amc = amcs.find(a => a.id === id);
    if (!amc) return;
    
    const techs = await fetchAPI('/technicians');
    const visitsDone = amc.visits_done || 0;
    const totalVisits = amc.visits_per_year || 4;

    }

    modal.className = 'modal';
    modal.style.display = 'flex';

          <div><strong>Client:</strong> ${amc.client_name}</div>
          <div><strong>Contract:</strong> ${amc.contract_type || 'Standard AMC'}</div>
          <div><strong>Visits Used:</strong> ${visitsDone} of ${totalVisits}</div>
        </div>
        
        <label>Visit Date *</label>
        <input type="date" id="visitDate" value="${new Date().toISOString().slice(0,10)}">
        
        <label>Technician *</label>
        <select id="visitTechnician">
          <option value="${amc.technician || ''}">${amc.technician || 'Select technician'}</option>
          ${techs.filter(t => t.active).map(t => `<option value="${t.name}">${t.name} — ${t.role || 'Technician'}</option>`).join('')}
        </select>
        
        <label>Tasks Completed *</label>
        <textarea id="visitTasks" rows="3" placeholder="e.g., AC filter cleaning, refrigerant top-up, electrical inspection..."></textarea>
        
        <label>Notes / Observations</label>
        <textarea id="visitNotes" rows="2" placeholder="Any issues found, recommendations, follow-up needed..."></textarea>
        
        <label>Hours Spent</label>
        <input type="number" id="visitHours" step="0.5" placeholder="e.g., 2.5">
        
        <div class="modal-actions">
          <button class="btn-secondary" id="cancelVisitBtn">Cancel</button>
          <button class="btn-primary" id="saveVisitBtn">Save Visit</button>
        </div>
      </div>
    `;
    
    document.body.appendChild(modal);
    
    document.getElementById('cancelVisitBtn').onclick = () => modal.remove();
    
    document.getElementById('saveVisitBtn').onclick = async () => {
      const visitDate = document.getElementById('visitDate').value;
      const technician = document.getElementById('visitTechnician').value;
      const tasksCompleted = document.getElementById('visitTasks').value;
      const notes = document.getElementById('visitNotes').value;
      const hoursSpent = parseFloat(document.getElementById('visitHours').value) || null;
      
      if (!visitDate) { alert('Please select visit date'); return; }
      if (!technician) { alert('Please select technician'); return; }
      if (!tasksCompleted) { alert('Please enter tasks completed'); return; }
      
      try {
        await fetchAPI('/amc-visits', {
          method: 'POST',
          body: JSON.stringify({
            amc_id: id,
            visit_date: visitDate,
            technician: technician,
            tasks_completed: tasksCompleted,
            notes: notes,
            hours_spent: hoursSpent
          })
        });
        
        const newVisitsDone = visitsDone + 1;
        await fetchAPI(`/amcs/${id}`, {
          method: 'PUT',
          body: JSON.stringify({
            visits_done: newVisitsDone,
            last_visit_date: visitDate,
            technician: technician
          })
        });
        
        alert(`✅ Visit logged successfully!\n\nVisit ${newVisitsDone} of ${totalVisits} completed.`);
        modal.remove();
        renderApp();
        
      } catch (err) {
        alert('Error saving visit: ' + err.message);
      }
    };
  };

      const startDate = document.getElementById('amcStart').value;
